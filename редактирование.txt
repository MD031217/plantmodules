// ===== ГЛОБАЛЬНЫЕ ПЕРЕМЕННЫЕ =====
File uploadFile;
String targetFilename;

// ===== ОБРАБОТЧИКИ =====
void handleProfile() {
  Serial.println("[HTTP] GET /profile");
  server.send_P(200, "text/html", PAGE_PROFILE);
}

void handleAdmin() {
  Serial.println("[HTTP] GET /admin");
  server.send_P(200, "text/html", PAGE_ADMIN);
}

void handleFileUpload() {
  HTTPUpload& upload = server.upload();
  if (upload.status == UPLOAD_FILE_START) {
    targetFilename = server.arg("filename");
    if (targetFilename.indexOf("..") != -1 || targetFilename.indexOf("\\") != -1) {
      Serial.println("[UPLOAD] Blocked: invalid filename");
      targetFilename = "";
      return;
    }
    targetFilename = "/img/" + targetFilename;
    Serial.println("[UPLOAD] Target: " + targetFilename);
    uploadFile = SD.open(targetFilename, FILE_WRITE);
  } else if (upload.status == UPLOAD_FILE_WRITE) {
    if (uploadFile) uploadFile.write(upload.buf, upload.currentSize);
  } else if (upload.status == UPLOAD_FILE_END) {
    if (uploadFile) {
      uploadFile.close();
      Serial.println("[UPLOAD] Saved: " + targetFilename + " (" + String(upload.totalSize) + " bytes)");
      server.send(200, "text/plain", "OK");
    } else {
      Serial.println("[UPLOAD] Failed: file handle error");
      server.send(500, "text/plain", "Write error");
    }
  }
}

void handleServeImage() {
  String filename = server.arg(0);
  if (filename.indexOf("..") != -1 || filename.indexOf("\\") != -1) {
    server.send(400, "text/plain", "Invalid path");
    return;
  }
  String fullPath = "/img/" + filename;
  if (!SD.exists(fullPath)) {
    server.send(404, "text/plain", "Image not found");
    return;
  }
  Serial.println("[HTTP] Serving: " + fullPath);
  File file = SD.open(fullPath, FILE_READ);
  String ct = "image/jpeg";
  String lower = filename; lower.toLowerCase();
  if (lower.endsWith(".png")) ct = "image/png";
  server.sendHeader("Content-Length", String(file.size()));
  server.sendHeader("Content-Type", ct);
  server.sendHeader("Cache-Control", "no-store, no-cache, must-revalidate");
  server.streamFile(file, ct);
  file.close();
}

void handleNotFound() {
  String uri = server.uri();
  Serial.println("[HTTP] 404 fallback: " + uri);
  
  // Проверяем, является ли запрос изображением (по расширению)
  if (uri.endsWith(".png") || uri.endsWith(".jpg") || uri.endsWith(".jpeg")) {
    // Извлекаем имя файла из пути (убираем ведущий /)
    String filename = uri;
    if (filename.startsWith("/")) filename = filename.substring(1);
    
    // Проверяем, есть ли файл в /img/
    String fullPath = "/img/" + filename;
    if (SD.exists(fullPath)) {
      Serial.println("[HTTP] Intercepted image: " + fullPath);
      File file = SD.open(fullPath, FILE_READ);
      String ct = "image/jpeg";
      String lower = filename; lower.toLowerCase();
      if (lower.endsWith(".png")) ct = "image/png";
      server.sendHeader("Content-Length", String(file.size()));
      server.sendHeader("Content-Type", ct);
      server.sendHeader("Cache-Control", "no-store, no-cache, must-revalidate");
      server.streamFile(file, ct);
      file.close();
      return;
    }
  }
  
  // Если не изображение или файл не найден — стандартная 404
  server.send(404, "text/plain", "Not found");
}

// ===== ИНИЦИАЛИЗАЦИЯ =====
void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n[SYSTEM] Booting...");

  Serial.print("[WIFI] Connecting to " + String(ssid));
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("\n[WIFI] Connected. IP: " + WiFi.localIP().toString());

  Serial.println("[SD] Initializing CS: " + String(SD_CS_PIN));
  if (!SD.begin(SD_CS_PIN, SPI, SPI_FREQ_INIT)) {
    Serial.println("[SD] 4MHz failed. Trying 1MHz...");
    if (!SD.begin(SD_CS_PIN, SPI, 1000000)) {
      Serial.println("[CRITICAL] SD init FAILED. Check wiring, format (FAT32/exFAT), power.");
      while(1) delay(1000);
    }
  }
  Serial.println("[SD] Mounted OK");
  uint8_t ct = SD.cardType();
  Serial.print("[SD] Type: ");
  if(ct==CARD_MMC) Serial.println("MMC"); else if(ct==CARD_SD) Serial.println("SDSC"); else if(ct==CARD_SDHC) Serial.println("SDHC/SDXC"); else Serial.println("UNKNOWN");
  Serial.print("[SD] Size: "); Serial.print(SD.cardSize()/(1024*1024)); Serial.println(" MB");

  if (!SD.exists("/img")) { SD.mkdir("/img"); Serial.println("[SD] Created /img"); }

  // Регистрация маршрутов (без regex!)
  server.on("/profile", HTTP_GET, handleProfile);
  server.on("/admin", HTTP_GET, handleAdmin);
  server.on("/upload", HTTP_POST, [](){}, handleFileUpload);
  server.on("/img", HTTP_GET, handleServeImage); // для /img?name=filename
  server.on("/", HTTP_GET, [](){ server.sendHeader("Location", "/profile"); server.send(302); });
  server.onNotFound(handleNotFound); // перехват всех остальных запросов

  server.begin();
  Serial.println("[HTTP] Server started on port 80");
  Serial.println("[INFO] Profile: http://" + WiFi.localIP().toString() + "/profile");
  Serial.println("[INFO] Admin: http://" + WiFi.localIP().toString() + "/admin");
}

void loop() {
  server.handleClient();
}
