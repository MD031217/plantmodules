void handleScan() {
    int n = WiFi.scanNetworks();
    String json = "[";
    for (int i = 0; i < n; ++i) {
        String ssid = WiFi.SSID(i);
        ssid.replace("\"", "\\\"");
        String enc = WiFi.encryptionType(i) == WIFI_AUTH_OPEN ? "OPEN" : "WPA2";
        json += "{\"ssid\":\"" + ssid + "\",\"encryption\":\"" + enc + "\"}";
        if (i < n - 1) json += ",";
    }
    json += "]";
    WiFi.scanDelete();
    server.send(200, "application/json", json);
}

void handleConfigure() {
    if (!server.hasArg("ssid")) {
        server.send(400, "application/json", "{\"status\":\"error\",\"message\":\"Отсутствует SSID\"}");
        return;
    }
    
    String ssid = server.arg("ssid");
    String pass = server.arg("pass");
    
    prefs.putString("ssid", ssid);
    prefs.putString("pass", pass);
    
    WiFi.disconnect(true);
    delay(500);

    if (pass.length() > 0) {
        WiFi.begin(ssid.c_str(), pass.c_str());
    } else {
        WiFi.begin(ssid.c_str());
    }
    
    int attempts = 0;
    while (WiFi.status() != WL_CONNECTED && attempts < 50) {
        delay(500);
        attempts++;
    }
    
    if (WiFi.status() == WL_CONNECTED) {
        String ip = WiFi.localIP().toString();
        server.send(200, "application/json", "{\"status\":\"success\",\"ip\":\"" + ip + "\",\"message\":\"Подключено!\"}");
    } else {
        server.send(200, "application/json", "{\"status\":\"error\",\"message\":\"Не удалось подключиться. Проверьте пароль.\"}");
    }
}

void setup() {
    Serial.begin(115200);
    prefs.begin("wifi-config", false);

    IPAddress local_IP(192, 168, 10, 1);
    IPAddress gateway(192, 168, 10, 1);
    IPAddress subnet(255, 255, 255, 0);
    
    WiFi.softAPConfig(local_IP, gateway, subnet); 
    WiFi.mode(WIFI_AP_STA);
    WiFi.softAP(apSSID, apPASS);
    
    Serial.println("[WiFi] AP started: " + String(apSSID) + " | IP: " + WiFi.softAPIP().toString());
    
    String s = prefs.getString("ssid", ""), p = prefs.getString("pass", "");
    if (s.length() > 0) {
        if (p.length() > 0) {
            WiFi.begin(s.c_str(), p.c_str());
        } else {
            WiFi.begin(s.c_str());
        }
        if (WiFi.waitForConnectResult(5000) == WL_CONNECTED) {
            Serial.println("[WiFi] STA connected: " + WiFi.localIP().toString());
        }
    }
    
    server.on("/settings.html", HTTP_GET, [](){
        server.send(200, "text/html", settings_html);
    });
    server.on("/api/scan", HTTP_GET, handleScan);
    server.on("/api/configure", HTTP_POST, handleConfigure);
    server.on("/", HTTP_GET, [](){
        server.sendHeader("Location", "/settings.html", true);
        server.send(302);
    });
    
    server.begin();
    Serial.println("[Setup] Web server started on port 80");
}
