// ===== ГЛОБАЛЬНЫЕ ПЕРЕМЕННЫЕ =====
File uploadFile;
String uploadPath;

// ===== ВСПОМОГАТЕЛЬНЫЕ ФУНКЦИИ =====
String protectPath(String folder, String filename) {
  filename.replace("../", "");
  filename.replace("..\\", "");
  filename.replace("\\", "/");
  if (filename.indexOf('/') != -1) {
    int last = filename.lastIndexOf('/');
    filename = filename.substring(last + 1);
  }
  // ✅ Явное преобразование в String для безопасной конкатенации
  return String("/") + folder + "/" + filename;
}

bool isValidFolder(String folder) {
  for (int i = 0; i < FOLDER_COUNT; i++) {
    if (folder == FOLDERS[i]) return true;
  }
  return false;
}

void ensureFolderExists(String folder) {
  String path = String("/") + folder;
  if (!SD.exists(path)) {
    SD.mkdir(path);
    Serial.println("[SD] Created: " + path);
  }
}

// ===== ОБРАБОТЧИКИ ЗАПРОСОВ =====
void handleRoot() {
  server.send_P(200, "text/html", index_html);
}

void handleFileUpload() {
  HTTPUpload& upload = server.upload();
  
  if (upload.status == UPLOAD_FILE_START) {
    String folder = server.arg("folder");
    if (!isValidFolder(folder)) folder = FOLDERS[0];
    
    String filename = upload.filename;
    uploadPath = protectPath(folder, filename);
    Serial.println("[UPLOAD] Saving: " + uploadPath + " (original: " + filename + ")");
    
    ensureFolderExists(folder);
    uploadFile = SD.open(uploadPath.c_str(), FILE_WRITE);
    
    if (!uploadFile) {
      Serial.println("[UPLOAD] ERROR: Cannot create file");
    }
  } 
  else if (upload.status == UPLOAD_FILE_WRITE) {
    if (uploadFile) {
      uploadFile.write(upload.buf, upload.currentSize);
    }
  } 
  else if (upload.status == UPLOAD_FILE_END) {
    if (uploadFile) {
      uploadFile.close();
      Serial.println("[UPLOAD] Done: " + uploadPath + " (" + String(upload.totalSize) + " B)");
    }
  }
}

void handleUploadResponse() {
  server.send(200, "text/plain", "OK");
}

void handleList() {
  String filter = server.arg("folder");
  String json = "{\"files\":[";
  bool first = true;
  int count = 0;
  const int MAX_FILES = 100;

  for (int f = 0; f < FOLDER_COUNT; f++) {
    String folder = FOLDERS[f];
    if (filter != "all" && filter != folder) continue;
    
    String dirPath = String("/") + folder;
    File dir = SD.open(dirPath.c_str());
    if (!dir || !dir.isDirectory()) { if (dir) dir.close(); continue; }
    
    while (count < MAX_FILES) {
      File entry = dir.openNextFile();
      if (!entry) break;
      
      if (!entry.isDirectory()) {
        String name = String(entry.name());
        if (!first) json += ",";
        
        String fullPath = String("/") + folder + "/" + name;
        json += "{\"name\":\"" + name + "\",";
        json += "\"folder\":\"" + folder + "\",";
        json += "\"path\":\"" + fullPath + "\"}";
        
        first = false;
        count++;
      }
      entry.close();
    }
    dir.close();
  }
  json += "]}";
  server.send(200, "application/json", json);
}

void handlePhoto() {
  String path = server.arg("path");
  if (!path.startsWith("/")) path = String("/") + path;
  
  if (path.indexOf("..") != -1) {
    server.send(400, "text/plain", "Invalid path");
    return;
  }
  if (!SD.exists(path)) {
    server.send(404, "text/plain", "Not Found");
    return;
  }

  File file = SD.open(path.c_str(), FILE_READ);
  if (!file) { server.send(500, "text/plain", "Error"); return; }
  
  String ct = "image/jpeg";
  String l = path; l.toLowerCase();
  if (l.endsWith(".png")) ct = "image/png";
  else if (l.endsWith(".gif")) ct = "image/gif";
  
  server.sendHeader("Content-Length", String(file.size()));
  server.sendHeader("Content-Type", ct);
  server.sendHeader("Cache-Control", "no-cache");
  server.streamFile(file, ct);
  file.close();
}

// ✅ ИСПРАВЛЕНА ОШИБКА КОМПИЛЯЦИИ ЗДЕСЬ
void handleDelete() {
  String path = server.arg("path");
  if (!path.startsWith("/")) path = String("/") + path;
  
  Serial.println("[HTTP] Delete requested: " + path);
  
  if (path.indexOf("..") != -1 || path.indexOf("\\") != -1) {
    server.send(400, "text/plain", "Invalid path");
    return;
  }
  
  // ✅ Исправлено: используем String() для конкатенации const char* массивов
  bool allowed = false;
  for (int i = 0; i < FOLDER_COUNT; i++) {
    String folderPath = String("/") + FOLDERS[i] + "/";
    if (path.startsWith(folderPath)) {
      allowed = true;
      break;
    }
  }
  if (!allowed) {
    server.send(403, "text/plain", "Access denied");
    return;
  }
  
  if (!SD.exists(path)) {
    server.send(404, "text/plain", "File not found");
    return;
  }

  if (SD.remove(path)) {
    Serial.println("[DELETE] Success: " + path);
    server.send(200, "text/plain", "OK");
  } else {
    Serial.println("[DELETE] Failed");
    server.send(500, "text/plain", "SD error");
  }
}

// ===== SETUP =====
void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n[BOOT] ESP32 starting...");

  WiFi.begin(ssid, password);
  Serial.print("[WiFi] Connecting");
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("\n[WiFi] OK: " + WiFi.localIP().toString());

  if (!SD.begin(SD_CS_PIN, SPI, SPI_FREQ_INIT)) {
    Serial.println("[SD] Retry at 1MHz...");
    if (!SD.begin(SD_CS_PIN, SPI, 1000000)) {
      Serial.println("[SD] FAILED!");
      while(1) delay(1000);
    }
  }
  Serial.println("[SD] OK");

  // Автоматическое создание папок при старте
  for (int i = 0; i < FOLDER_COUNT; i++) {
    ensureFolderExists(FOLDERS[i]);
  }

  server.on("/", HTTP_GET, handleRoot);
  server.on("/upload", HTTP_POST, handleUploadResponse, handleFileUpload);
  server.on("/list", HTTP_GET, handleList);
  server.on("/photo", HTTP_GET, handlePhoto);
  server.on("/delete", HTTP_DELETE, handleDelete);

  server.begin();
  Serial.println("[HTTP] Ready: http://" + WiFi.localIP().toString());
}

void loop() {
  server.handleClient();
}
