// ===== ГЛОБАЛЬНЫЕ ПЕРЕМЕННЫЕ ДЛЯ ЗАГРУЗКИ =====
File uploadFile;
String uploadFilename;

// ===== ОБРАБОТЧИКИ ЗАПРОСОВ =====
void handleRoot() {
  Serial.println("[HTTP] Client requested main page");
  server.send_P(200, "text/html", index_html);
}

void handleFileUpload() {
  HTTPUpload& upload = server.upload();
  
  if (upload.status == UPLOAD_FILE_START) {
    uploadFilename = "/" + upload.filename;
    uploadFilename.replace("../", ""); // Защита от path traversal
    if (!uploadFilename.endsWith(".jpg") && !uploadFilename.endsWith(".png") && !uploadFilename.endsWith(".jpeg")) {
      uploadFilename += ".jpg";
    }
    Serial.println("[UPLOAD] Starting: " + uploadFilename);
    uploadFile = SD.open(uploadFilename, FILE_WRITE);
  } 
  else if (upload.status == UPLOAD_FILE_WRITE) {
    if (uploadFile) {
      uploadFile.write(upload.buf, upload.currentSize);
    }
  } 
  else if (upload.status == UPLOAD_FILE_END) {
    if (uploadFile) {
      uploadFile.close();
      Serial.println("[UPLOAD] Finished: " + uploadFilename + " (" + String(upload.totalSize) + " bytes)");
    } else {
      Serial.println("[UPLOAD] Failed: File handle closed unexpectedly");
    }
  }
}

void handleUploadResponse() {
  server.send(200, "text/plain", "OK");
}

void handleList() {
  Serial.println("[HTTP] Client requested file list");
  File root = SD.open("/");
  String json = "[";
  bool first = true;
  int count = 0;
  const int MAX_FILES = 50; // Ограничение для экономии памяти

  if (!root) {
    server.send(500, "text/plain", "Failed to open SD root");
    return;
  }

  while (count < MAX_FILES) {
    File entry = root.openNextFile();
    if (!entry) break;
    
    if (!entry.isDirectory()) {
      String name = String(entry.name());
      String lower = name;
      lower.toLowerCase();
      if (lower.endsWith(".jpg") || lower.endsWith(".jpeg") || lower.endsWith(".png")) {
        if (!first) json += ",";
        // Убираем начальный /, если он есть
        if (name.startsWith("/")) name = name.substring(1);
        json += "\"" + name + "\"";
        first = false;
        count++;
      }
    }
    entry.close();
  }
  json += "]";
  root.close();
  
  server.send(200, "application/json", json);
}

void handlePhoto() {
  String name = "/" + server.arg("name");
  Serial.println("[HTTP] Client requested photo: " + name);
  
  if (!SD.exists(name)) {
    server.send(404, "text/plain", "Not Found");
    return;
  }

  File file = SD.open(name, FILE_READ);
  String contentType = "image/jpeg";
  String lower = name;
  lower.toLowerCase();
  if (lower.endsWith(".png")) contentType = "image/png";

  server.sendHeader("Content-Length", String(file.size()));
  server.sendHeader("Content-Type", contentType);
  server.sendHeader("Cache-Control", "no-cache, no-store, must-revalidate");
  server.streamFile(file, contentType);
  file.close();
}

void handleDelete() {
  String name = "/" + server.arg("name");
  Serial.println("[HTTP] Client requested delete: " + name);
  
  if (name.indexOf("..") != -1 || name.indexOf("\\") != -1) {
    Serial.println("[DELETE] Blocked: Invalid path characters");
    server.send(400, "text/plain", "Invalid filename");
    return;
  }
  
  if (!SD.exists(name)) {
    Serial.println("[DELETE] Failed: File not found on SD");
    server.send(404, "text/plain", "File not found");
    return;
  }

  if (SD.remove(name)) {
    Serial.println("[DELETE] Success: " + name);
    server.send(200, "text/plain", "OK");
  } else {
    Serial.println("[DELETE] Failed: SD.remove() error");
    server.send(500, "text/plain", "SD write error");
  }
}

// ===== ИНИЦИАЛИЗАЦИЯ И НАСТРОЙКА =====
void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n[SYSTEM] ESP32 Booting...");

  // 1. Подключение к Wi-Fi
  Serial.print("[WIFI] Connecting to: " + String(ssid));
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\n[WIFI] Connected. IP: " + WiFi.localIP().toString());

  // 2. Надёжная инициализация SD-карты
  Serial.println("[SD] Initializing with CS pin: " + String(SD_CS_PIN));
  if (!SD.begin(SD_CS_PIN, SPI, SPI_FREQ_INIT)) {
    Serial.println("[SD] Mount failed at " + String(SPI_FREQ_INIT/1000000) + " MHz. Trying fallback 1 MHz...");
    if (!SD.begin(SD_CS_PIN, SPI, 1000000)) {
      Serial.println("[CRITICAL] SD card initialization FAILED.");
      Serial.println("Checklist:");
      Serial.println("1. Verify SD_CS_PIN matches your hardware wiring");
      Serial.println("2. Format card to FAT32 (for <=32GB) or exFAT (for >32GB)");
      Serial.println("3. Check power supply (3.3V stable) and wiring length (<15cm)");
      Serial.println("4. Ensure module has logic level shifter if powered by 5V");
      while (1) { delay(1000); }
    }
  }
  Serial.println("[SD] Mount successful");

  // Диагностика карты
  uint8_t cardType = SD.cardType();
  Serial.print("[SD] Card Type: ");
  switch (cardType) {
    case CARD_MMC:  Serial.println("MMC"); break;
    case CARD_SD:   Serial.println("SDSC (<=2GB)"); break;
    case CARD_SDHC: Serial.println("SDHC/SDXC (>2GB)"); break;
    default:        Serial.println("UNKNOWN"); break;
  }
  Serial.print("[SD] Capacity: "); Serial.print(SD.cardSize() / (1024 * 1024)); Serial.println(" MB");
  Serial.print("[SD] Total space: "); Serial.print(SD.totalBytes() / (1024 * 1024)); Serial.println(" MB");
  Serial.print("[SD] Used space: "); Serial.print(SD.usedBytes() / (1024 * 1024)); Serial.println(" MB");

  // 3. Регистрация HTTP-маршрутов
  server.on("/", HTTP_GET, handleRoot);
  server.on("/upload", HTTP_POST, handleUploadResponse, handleFileUpload);
  server.on("/list", HTTP_GET, handleList);
  server.on("/photo", HTTP_GET, handlePhoto);
  server.on("/delete", HTTP_DELETE, handleDelete);

  server.begin();
  Serial.println("[HTTP] Web server started on port 80");
  Serial.println("[INFO] Open http://" + WiFi.localIP().toString() + " in browser");
}

void loop() {
  server.handleClient();
}
