File uploadFile;
String uploadFilename;

// ===== ОБРАБОТЧИКИ ЗАПРОСОВ =====
void handleRoot() {
  Serial.println("[HTTP] Admin page requested");
  server.send_P(200, "text/html", PAGE_ADMIN);
}

void handleProfile() {
  Serial.println("[HTTP] Profile page requested");
  server.send_P(200, "text/html", PAGE_PROFILE);
}

// Загрузка админ-фото в /gallery/
void handleAdminUpload() {
  HTTPUpload& upload = server.upload();
  if (upload.status == UPLOAD_FILE_START) {
    uploadFilename = "/gallery/" + upload.filename;
    uploadFilename.replace("../", "");
    if (!uploadFilename.endsWith(".jpg") && !uploadFilename.endsWith(".png") && !uploadFilename.endsWith(".jpeg")) uploadFilename += ".jpg";
    Serial.println("[ADMIN] Upload started: " + uploadFilename);
    uploadFile = SD.open(uploadFilename, FILE_WRITE);
  } else if (upload.status == UPLOAD_FILE_WRITE) {
    if (uploadFile) uploadFile.write(upload.buf, upload.currentSize);
  } else if (upload.status == UPLOAD_FILE_END) {
    if (uploadFile) {
      uploadFile.close();
      Serial.println("[ADMIN] Upload finished: " + uploadFilename + " (" + String(upload.totalSize) + " bytes)");
    }
  }
}
void handleAdminResponse() { server.send(200, "text/plain", "OK"); }

// Загрузка фото профиля (только /profile.jpg)
void handleProfileUpload() {
  HTTPUpload& upload = server.upload();
  if (upload.status == UPLOAD_FILE_START) {
    uploadFilename = "/profile.jpg";
    Serial.println("[PROFILE] Profile photo upload started");
    uploadFile = SD.open(uploadFilename, FILE_WRITE);
  } else if (upload.status == UPLOAD_FILE_WRITE) {
    if (uploadFile) uploadFile.write(upload.buf, upload.currentSize);
  } else if (upload.status == UPLOAD_FILE_END) {
    if (uploadFile) {
      uploadFile.close();
      Serial.println("[PROFILE] Profile photo updated successfully");
    }
  }
}
void handleProfileResponse() { server.send(200, "text/plain", "OK"); }

// Список только админ-фото
void handleListGallery() {
  Serial.println("[HTTP] Gallery list requested");
  File root = SD.open("/gallery");
  if (!root) { server.send(500, "text/plain", "Failed to open gallery dir"); return; }
  String json = "[";
  bool first = true;
  int count = 0;
  while (count < 50) {
    File entry = root.openNextFile();
    if (!entry) break;
    if (!entry.isDirectory()) {
      String name = String(entry.name());
      String lower = name; lower.toLowerCase();
      if (lower.endsWith(".jpg") || lower.endsWith(".jpeg") || lower.endsWith(".png")) {
        if (!first) json += ",";
        json += "\"" + name + "\"";
        first = false;
        count++;
      }
    }
    entry.close();
  }
  json += "]";
  root.close();
  server.send(200, "application/json", json);
}

// Отдача изображений с защитой от traversal
void handlePhoto() {
  String name = server.arg("name");
  if (name.indexOf("..") != -1 || name.indexOf("\\") != -1) {
    server.send(400, "text/plain", "Invalid path");
    return;
  }
  // Разрешаем только /gallery/* и /profile.jpg
  bool allowed = (name.startsWith("gallery/") || name == "profile.jpg");
  String fullPath = "/" + name;
  if (!allowed || !SD.exists(fullPath)) {
    server.send(404, "text/plain", "Not found or access denied");
    return;
  }
  Serial.println("[HTTP] Photo requested: " + fullPath);
  File file = SD.open(fullPath, FILE_READ);
  String ct = "image/jpeg";
  String lower = fullPath; lower.toLowerCase();
  if (lower.endsWith(".png")) ct = "image/png";
  server.sendHeader("Content-Length", String(file.size()));
  server.sendHeader("Content-Type", ct);
  server.sendHeader("Cache-Control", "no-cache, must-revalidate");
  server.streamFile(file, ct);
  file.close();
}

// Удаление только из /gallery/
void handleDeleteAdmin() {
  String name = server.arg("name");
  if (name.indexOf("..") != -1 || name.indexOf("\\") != -1) {
    Serial.println("[ADMIN] Delete blocked: invalid path");
    server.send(400, "text/plain", "Invalid filename");
    return;
  }
  String fullPath = "/gallery/" + name;
  Serial.println("[ADMIN] Delete requested: " + fullPath);
  if (!SD.exists(fullPath)) {
    Serial.println("[ADMIN] Delete failed: not found");
    server.send(404, "text/plain", "File not found");
    return;
  }
  if (SD.remove(fullPath)) {
    Serial.println("[ADMIN] Deleted successfully: " + fullPath);
    server.send(200, "text/plain", "OK");
  } else {
    Serial.println("[ADMIN] Delete failed: SD error");
    server.send(500, "text/plain", "SD write error");
  }
}

// ===== ИНИЦИАЛИЗАЦИЯ =====
void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n[SYSTEM] ESP32 Booting...");

  // Wi-Fi
  Serial.print("[WIFI] Connecting to: " + String(ssid));
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("\n[WIFI] Connected. IP: " + WiFi.localIP().toString());

  // SD Инициализация (надёжная, с fallback)
  Serial.println("[SD] Initializing with CS pin: " + String(SD_CS_PIN));
  if (!SD.begin(SD_CS_PIN, SPI, SPI_FREQ_INIT)) {
    Serial.println("[SD] Mount failed at 4 MHz. Trying 1 MHz fallback...");
    if (!SD.begin(SD_CS_PIN, SPI, 1000000)) {
      Serial.println("[CRITICAL] SD card initialization FAILED.");
      Serial.println("Checklist:");
      Serial.println("1. Verify SD_CS_PIN matches hardware");
      Serial.println("2. Format card to FAT32 (<=32GB) or exFAT (>32GB)");
      Serial.println("3. Check 3.3V power and wiring (<15cm)");
      while(1) delay(1000);
    }
  }
  Serial.println("[SD] Mount successful");
  uint8_t ct = SD.cardType();
  Serial.print("[SD] Type: ");
  if(ct==CARD_MMC) Serial.println("MMC"); else if(ct==CARD_SD) Serial.println("SDSC"); else if(ct==CARD_SDHC) Serial.println("SDHC/SDXC"); else Serial.println("UNKNOWN");
  Serial.print("[SD] Capacity: "); Serial.print(SD.cardSize()/(1024*1024)); Serial.println(" MB");

  // Создание директорий
  if (!SD.exists("/gallery")) {
    SD.mkdir("/gallery");
    Serial.println("[SD] Created /gallery directory");
  }

  // Регистрация маршрутов
  server.on("/", HTTP_GET, handleRoot);
  server.on("/profile", HTTP_GET, handleProfile);
  server.on("/upload_admin", HTTP_POST, handleAdminResponse, handleAdminUpload);
  server.on("/upload_profile", HTTP_POST, handleProfileResponse, handleProfileUpload);
  server.on("/list_gallery", HTTP_GET, handleListGallery);
  server.on("/photo", HTTP_GET, handlePhoto);
  server.on("/delete_admin", HTTP_DELETE, handleDeleteAdmin);

  server.begin();
  Serial.println("[HTTP] Server started on port 80");
  Serial.println("[INFO] Admin: http://" + WiFi.localIP().toString());
  Serial.println("[INFO] Profile: http://" + WiFi.localIP().toString() + "/profile");
}

void loop() {
  server.handleClient();
}
