File uploadFile;
String uploadFilename;

void handleRoot() {
  Serial.println("[INFO] Client requested main page");
  server.send_P(200, "text/html", index_html);
}

void handleFileUpload() {
  HTTPUpload& upload = server.upload();
  
  if (upload.status == UPLOAD_FILE_START) {
    uploadFilename = "/" + upload.filename;
    uploadFilename.replace("../", "");
    if (!uploadFilename.endsWith(".jpg") && !uploadFilename.endsWith(".png")) {
      uploadFilename += ".jpg";
    }
    Serial.println("[UPLOAD] Started: " + uploadFilename);
    uploadFile = SD.open(uploadFilename, FILE_WRITE);
  } 
  else if (upload.status == UPLOAD_FILE_WRITE) {
    if (uploadFile) uploadFile.write(upload.buf, upload.currentSize);
  } 
  else if (upload.status == UPLOAD_FILE_END) {
    if (uploadFile) {
      uploadFile.close();
      Serial.println("[UPLOAD] Completed: " + uploadFilename + " (" + String(upload.totalSize) + " bytes)");
    }
  }
}

void handleUploadResponse() {
  server.send(200, "text/plain", "OK");
}

void handleList() {
  Serial.println("[LIST] Client requested file list");
  File root = SD.open("/");
  String json = "[";
  bool first = true;
  int count = 0;

  while (count < 100) {
    File entry = root.openNextFile();
    if (!entry) break;
    
    if (!entry.isDirectory()) {
      String name = String(entry.name());
      if (name.endsWith(".jpg") || name.endsWith(".png")) {
        if (!first) json += ",";
        json += "\"" + name.substring(1) + "\"";
        first = false;
        count++;
      }
    }
    entry.close();
  }
  json += "]";
  root.close();
  
  server.send(200, "application/json", json);
}

void handlePhoto() {
  String name = "/" + server.arg("name");
  Serial.println("[PHOTO] Client requested: " + name);
  
  if (!SD.exists(name)) {
    server.send(404, "text/plain", "Not Found");
    return;
  }

  File file = SD.open(name, FILE_READ);
  String contentType = "image/jpeg";
  if (name.endsWith(".png")) contentType = "image/png";

  server.sendHeader("Content-Length", String(file.size()));
  server.sendHeader("Content-Type", contentType);
  server.streamFile(file, contentType);
  file.close();
}

void handleDelete() {
  String name = "/" + server.arg("name");
  Serial.println("[DELETE] Client requested: " + name);
  
  if (name.indexOf("..") != -1) {
    Serial.println("[DELETE] Blocked: path traversal attempt");
    server.send(400, "text/plain", "Invalid path");
    return;
  }
  
  if (!SD.exists(name)) {
    Serial.println("[DELETE] Failed: file not found");
    server.send(404, "text/plain", "File not found");
    return;
  }

  if (SD.remove(name)) {
    Serial.println("[DELETE] Success: " + name);
    server.send(200, "text/plain", "OK");
  } else {
    Serial.println("[DELETE] Failed: SD.remove() returned false");
    server.send(500, "text/plain", "Delete failed");
  }
}

void setup() {
  Serial.begin(115200);
  Serial.println("Booting...");
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi connected. IP: " + WiFi.localIP().toString());

  if (!SD.begin(SD_CS_PIN)) {
    Serial.println("[ERROR] SD card initialization failed");
    while(1);
  }
  Serial.println("[INFO] SD card initialized");

  server.on("/", HTTP_GET, handleRoot);
  server.on("/upload", HTTP_POST, handleUploadResponse, handleFileUpload);
  server.on("/list", HTTP_GET, handleList);
  server.on("/photo", HTTP_GET, handlePhoto);
  server.on("/delete", HTTP_DELETE, handleDelete);

  server.begin();
  Serial.println("[INFO] HTTP server started on port 80");
}

void loop() {
  server.handleClient();
}
