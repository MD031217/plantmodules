void handleScan() {
    int n = WiFi.scanNetworks();
    String json = "[";
    for (int i = 0; i < n; ++i) {
        String ssid = WiFi.SSID(i);
        ssid.replace("\"", "\\\""); // Экранирование кавычек для JSON
        json += "{\"ssid\":\"" + ssid + "\",\"rssi\":" + String(WiFi.RSSI(i)) + "}";
        if (i < n - 1) json += ",";
    }
    json += "]";
    WiFi.scanDelete();
    server.send(200, "application/json", json);
}

void handleConfigure() {
    if (!server.hasArg("ssid") || !server.hasArg("pass")) {
        server.send(400, "application/json", "{\"status\":\"error\",\"message\":\"Отсутствуют данные\"}");
        return;
    }

    String ssid = server.arg("ssid");
    String pass = server.arg("pass");
    
    // Сохраняем в NVS (Preferences)
    prefs.putString("ssid", ssid);
    prefs.putString("pass", pass);

    WiFi.disconnect(true);
    delay(500);
    WiFi.begin(ssid.c_str(), pass.c_str());

    // Ждём подключения до 15 секунд
    int attempts = 0;
    while (WiFi.status() != WL_CONNECTED && attempts < 30) {
        delay(500);
        attempts++;
    }

    if (WiFi.status() == WL_CONNECTED) {
        String ip = WiFi.localIP().toString();
        server.send(200, "application/json", "{\"status\":\"success\",\"ip\":\"" + ip + "\",\"message\":\"Подключено к сети!\"}");
    } else {
        server.send(200, "application/json", "{\"status\":\"error\",\"message\":\"Не удалось подключиться. Проверьте пароль или покрытие.\"}");
    }
}

void setup() {
    Serial.begin(115200);
    prefs.begin("wifi-config", false);

    // Режим AP + STA одновременно: не рвёт соединение с клиентом во время переподключения
    WiFi.mode(WIFI_AP_STA);
    WiFi.softAP(apSSID, apPASS);
    Serial.print("AP запущен. IP: "); Serial.println(WiFi.softAPIP());

    // Попытка автоподключения к последней сохранённой сети
    String savedSSID = prefs.getString("ssid", "");
    String savedPASS = prefs.getString("pass", "");
    if (savedSSID.length() > 0) {
        WiFi.begin(savedSSID.c_str(), savedPASS.c_str());
        if (WiFi.waitForConnectResult(5000) == WL_CONNECTED) {
            Serial.print("Автоподключение успешно. STA IP: "); Serial.println(WiFi.localIP());
        }
    }

    // Роуты
    server.on("/", HTTP_GET, [](){
        server.sendHeader("Location", "/settings.html", true);
        server.send(302);
    });
    server.on("/settings.html", HTTP_GET, [](){
        server.send(200, "text/html", settings_html);
    });
    server.on("/api/scan", HTTP_GET, handleScan);
    server.on("/api/configure", HTTP_POST, handleConfigure);

    server.begin();
    Serial.println("Веб-сервер запущен на порту 80");
}

void loop() {
    server.handleClient();
}
